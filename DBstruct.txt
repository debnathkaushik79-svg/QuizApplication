CREATE DATABASE QuizApplication;

USE QuizApplication;


CREATE TABLE Users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);


CREATE TABLE Questions (
    question_id INT AUTO_INCREMENT PRIMARY KEY,
    topic VARCHAR(30),
    question TEXT,
    option1 VARCHAR(255),
    option2 VARCHAR(255),
    option3 VARCHAR(255),
    option4 VARCHAR(255),
    answer VARCHAR(255)
);


CREATE TABLE Scores (
    score_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    topic VARCHAR(30),
    score INT,
    total INT,
    percentage DOUBLE,
    quiz_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(user_id) REFERENCES Users(user_id)
);

Java Quiz Application

Developed a desktop-based Quiz Application using Java Swing and JDBC.
Implemented user registration, multiple quiz categories, randomized questions, score calculation, and result evaluation.
Integrated MySQL database to manage users, questions, and quiz scores.
Applied Object-Oriented Programming principles and DAO architecture for modular and maintainable code.

Technologies: Java, Swing, JDBC, MySQL, SQL, OOP
